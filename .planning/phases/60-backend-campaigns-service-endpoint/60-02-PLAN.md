    ```graphql
    query {
      campaigns(first: 5) {
        edges {
          node {
            id
            name
            objective
            first_seen
            last_seen
            attributed_to: stixCoreRelationships(relationship_type: "attributed-to", toTypes: ["Intrusion-Set"], first: 10) {
              edges { node { to { ... on IntrusionSet { id name } } } }
            }
          }
        }
      }
    }
    ```
